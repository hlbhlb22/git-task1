4. 
DISK1="/dev/sdb"
DISK2="/dev/sdc"
VG_NAME="myvg"

sudo pvcreate $DISK1

sudo vgcreate $VG_NAME $DISK1

sudo pvs
sudo vgs

5. 
VG_SIZE=$(sudo vgs $VG_NAME --units g --noheadings --nosuffix -o vg_free)
LV_SIZE=$(echo "$VG_SIZE / 2" | bc)

sudo lvcreate -L ${LV_SIZE}G -n lv1 $VG_NAME
sudo lvcreate -l 100%FREE -n lv2 $VG_NAME  # Use remaining space

sudo lvs

6. 

sudo mkfs.xfs /dev/$VG_NAME/lv1
sudo mkfs.xfs /dev/$VG_NAME/lv2

sudo blkid | grep $VG_NAME

7. 
sudo mkdir -p /mnt/lv1
sudo mkdir -p /mnt/lv2

sudo mount /dev/$VG_NAME/lv1 /mnt/lv1
sudo mount /dev/$VG_NAME/lv2 /mnt/lv2

mount | grep $VG_NAME 

8.
df -h > after_lv_creation_df.txt
echo "=== After LV Creation (df -h) ==="
cat after_lv_creation_df.txt

9. 
sudo pvcreate $DISK2

sudo vgextend $VG_NAME $DISK2

10. 
sudo vgs $VG_NAME

DISK2_SIZE=$(sudo pvs $DISK2 --units g --noheadings --nosuffix -o pv_size)

sudo lvextend -L +${DISK2_SIZE}G /dev/$VG_NAME/lv1

sudo lvs $VG_NAME

11.
