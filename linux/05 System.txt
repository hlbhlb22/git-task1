#!/bin/bash
# /usr/local/bin/system_monitor.sh - Updated version

LOG_FILE="/var/log/system_monitor.log"
OVERLOAD_FILE="/var/log/overload"

while true; do
    # Get uptime output
    UPTIME_OUTPUT=$(uptime)
    TIMESTAMP=$(date "+%Y-%m-%d %H:%M:%S")
    
    # Write to log file
    echo "$TIMESTAMP: $UPTIME_OUTPUT" >> "$LOG_FILE"
    
    # Extract 1-minute load average
    LOAD_AVG=$(echo "$UPTIME_OUTPUT" | awk -F'[ ,:]+' '{print $(NF-2)}' | tr -d ',')
    
    # Check if load average > 1
    if (( $(echo "$LOAD_AVG > 1" | bc -l) )); then
        echo "$TIMESTAMP: LOAD WARNING - $UPTIME_OUTPUT" >> "$OVERLOAD_FILE"
    fi
    
    # Sleep for 15 seconds
    sleep 15
done


sudo systemctl restart system-monitor.service


sudo tail -f /var/log/system_monitor.log

#!/bin/bash
# /usr/local/bin/cleanup_monitor.sh

OVERLOAD_FILE="/var/log/overload"
CLEANUP_LOG="/var/log/cleanup.log"
MAX_SIZE_KB=50

# Check if overload file exists
if [[ -f "$OVERLOAD_FILE" ]]; then
    # Get file size in kilobytes
    FILE_SIZE=$(du -k "$OVERLOAD_FILE" | cut -f1)
    
    if [[ $FILE_SIZE -gt $MAX_SIZE_KB ]]; then
        # Log cleanup action
        TIMESTAMP=$(date "+%Y-%m-%d %H:%M:%S")
        echo "$TIMESTAMP: Cleaned $OVERLOAD_FILE (size: ${FILE_SIZE}KB)" >> "$CLEANUP_LOG"
        
        # Clear the file
        > "$OVERLOAD_FILE"
        
        echo "Cleaned overload file at $TIMESTAMP"
    else
        echo "File size ($FILE_SIZE KB) within limits"
    fi
else
    echo "Overload file does not exist"
fi

sudo chmod +x /usr/local/bin/cleanup_monitor.sh

#!/bin/bash
# /usr/local/bin/check_monitor_status.sh

SERVICE="system-monitor.service"
STATUS_LOG="/var/log/service_status.log"

# Check service status
SERVICE_STATUS=$(systemctl is-active "$SERVICE")
TIMESTAMP=$(date "+%Y-%m-%d %H:%M:%S")

# Log status
echo "$TIMESTAMP: $SERVICE is $SERVICE_STATUS" >> "$STATUS_LOG"

# Restart if not active
if [[ "$SERVICE_STATUS" != "active" ]]; then
    echo "$TIMESTAMP: Restarting $SERVICE" >> "$STATUS_LOG"
    systemctl restart "$SERVICE"
fi
