sudo nano /home/unty/gocalc/docker-compose.yml

version: '3'

services:
  db:
    image: postgres:10
    networks:
      - postgres
    ports:
      - "5432:5432"
    environment:
      POSTGRES_DB: 'postgres'
      POSTGRES_USER: 'user'
      POSTGRES_PASSWORD: 'DatabasePassword'

  backend:
    image: gocalc:5555
    networks:
      - nginx
      - postgres
    environment:
      - POSTGRES_URI=postgres://user:DatabasePassword@db/postgres?sslmode=disable
      - LISTEN_ADDRESS=:7000
    restart: unless-stopped
    depends_on:
      - db

  nginx:
    image: nginx:latest
    networks:
      - nginx
    ports:
      - "8089:80"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
    depends_on:
      - db
      - backend

networks:
  nginx:
  postgres:

sudo docker-compose -p inno28 up -d
