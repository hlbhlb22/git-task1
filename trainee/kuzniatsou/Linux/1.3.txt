#!/bin/bash

OVERLOAD_LOG="/var/log/overload.log"
UPTIME_LOG="/var/log/uptime.log"
CLEANUP_LOG="/var/log/cleanup.log"

while true; do
    uptime_output=$(uptime)
    load_average=$(echo $uptime_output | awk '{print $(NF-2)}' | sed 's/,//')

    # Проверяем, превышает ли Load Average за минуту значение 1.0
    if (( $(echo "$load_average > 1.0" | bc -l) )); then
        echo $uptime_output >> $OVERLOAD_LOG
    else
        echo $uptime_output >> $UPTIME_LOG
    fi

    # Проверяем размер файла overload.log и очищаем, если превышает 50 KB
    if [ $(stat -c%s "$OVERLOAD_LOG") -ge 51200 ]; then
        echo "$(date): overload.log очищен" >> $CLEANUP_LOG
        > $OVERLOAD_LOG
    fi
    
    sleep 15
done
