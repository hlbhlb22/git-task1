# tasks file for install_nginx
- name: Check if NodeJS directory exists
  stat:
     path: "{{nodejspath}}"
  register: nodejs_info

- name: Message if directory doesn't exist
  debug:
    var: nodejs_info
  when: not nodejs_info.stat.exists

- name: Ensure the directory exists
  file:
    path: "{{nodejspath}}"
    state: directory
    mode: '0755'
  delegate_to: localhost

- name: Download NodeJS tarball
  get_url:
    url: "https://nodejs.org/dist/{{nodejsversion}}/node-{{nodejsversion}}-linux-x64.tar.xz"
    dest: "{{nodejspath}}/node-{{nodejsversion}}-linux-x64.tar.xz"
    mode: '0755'
  delegate_to: localhost

- name: Unarchive NodeJS tarball
  unarchive:
    src: "{{nodejspath}}/node-{{nodejsversion}}-linux-x64.tar.xz"
    dest: "{{nodejspath}}" 
  delegate_to: localhost

- name: Delete the tarball file
  file:
    path: "{{nodejspath}}/node-{{nodejsversion}}-linux-x64.tar.xz"
    state: absent
  delegate_to: localhost

- name: Create dir for mv
  file:
     path: "{{dirformv}}"
     state: directory 

- name: Move extracted NodeJS to the installation directory
  command: "mv {{nodejspath}}/node-{{nodejsversion}}-linux-x64 {{dirformv}}"
  delegate_to: localhost

- name: Remove directory for installation locally
  file:
    path: "{{nodejspath}}"
    state: absent
  delegate_to: localhost
  run_once: true
     
- name: Create a symbolic link
  file:
    src: "{{dirformv}}"
    path: "{{symboliclink}}"
    state: link

# tasks file for git_clone_site
- name: Create rep for clone
  file:
    path: /usr/local/lib/git/nodejs-ansible
    state: directory
  when: epic == true

- name: Clone repo
  git: 
    repo: https://devops-gitlab.inno.ws/devops-board/nodejs-ansible.git
    dest: /usr/local/lib/git/nodejs-ansible
    version: master
    recursive: true
    force: yes
    depth: 1
  when: epic == true

- name: Install NodeJS, npm
  apt:
    name:
      - nodejs
      - npm
        state: present
        update_cache: yes
  when: epic == true
    
- name: Download packages from package.json
  npm:
    path: /usr/local/lib/git/nodejs-ansible
  when: epic == true
 
- name: Wait
  wait_for:
     host: 127.0.0.1
     port: 3000
     delay: 10
     timeout: 20
  when: epic == true
 
- name: Check works
  uri:
    url: http://127.0.0.1:3000
    method: GET
    return_content: true
  register: local_info
  when: epic == true
