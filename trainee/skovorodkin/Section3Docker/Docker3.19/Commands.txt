FROM ubuntu:18.04

#Обновляем список пакетов и установливаем nginx
RUN apt-get update&&apt-get install -y nginx

#Копируем конфигурационный файл в контейнер
COPY nginx.conf /etc/nginx/nginx.conf

#Задаем раб.директорию для следующих инструкций
WORKDIR /etc/nginx/

#Запускаем nginx при запуске контейнера
ENTRYPOINT ["nginx"]

#Параметры для запуска nginx при запуске контейнера
CMD ["-g","daemon off;"]

#Указываем директорию для постоянного хранения файлов
VOLUME /var/ling/nginx

#Указываем порт, который планируем открыть для того, чтобы связаться с
#контейнром 

EXPOSE 80
#

sudo docker build -t nginx:inno-dkr-09 /home/unty/Docker3.19
