    worker_connections 1024;
}

    server {
        listen 80;
        server_name localhost;

            index index.html;
        }
    }

cat > docker-compose-mounted.yml << 'EOF'
version: '3.8'

services:
  nginx:
    image: nginx:stable
    container_name: nginx
    ports:
      - "127.0.0.1:18888:80"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
EOF
