CREATE DATABASE TEST;
USE TEST;
CREATE TABLE master (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL
);
CREATE TABLE details (
    id INT PRIMARY KEY AUTO_INCREMENT,
    master_id INT,
    detail_info VARCHAR(255),
    FOREIGN KEY (master_id) REFERENCES master(id)
);
